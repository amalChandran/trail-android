@Composable fun RuntimeColorExample(brandColor: TrailColor) {
    val effect = remember(brandColor) {
        trailEffect {
            stroke(brandColor, width = 6.0)
            reveal(duration = 2.seconds)
        }
    }
    TrailCanvas(examplePath, Modifier.fillMaxSize(), effect = effect)
}
