    let brandColor: TrailColor
    var body: some View {
        let effect = TrailEffect {
            Style(MetroStyle(color: brandColor))
            Animate(TrailAnimations.custom(QuadraticReveal(), duration: .seconds(2)))
        }
        TrailCanvas(path: ExampleGeometry.path, effect: effect)
    }
}
